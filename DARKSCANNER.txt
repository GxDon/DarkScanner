#!/usr/bin/env python3

import argparse
import json
import socket


def scan(host, start, end, timeout):
    results = []

    for port in range(start, end + 1):
        with socket.socket(
            socket.AF_INET,
            socket.SOCK_STREAM
        ) as sock:

            sock.settimeout(timeout)

            if sock.connect_ex((host, port)) == 0:
                results.append(port)
                print(f"[OPEN] {port}")

    return results


def main():
    parser = argparse.ArgumentParser(
        description="DarkScanner - escaneo TCP autorizado"
    )

    parser.add_argument("target")
    parser.add_argument("--start", type=int, default=1)
    parser.add_argument("--end", type=int, default=1024)
    parser.add_argument("--timeout", type=float, default=0.5)
    parser.add_argument(
        "-o",
        "--output",
        default="darkscanner_report.json"
    )

    args = parser.parse_args()

    if not 1 <= args.start <= 65535:
        raise SystemExit("[!] Puerto inicial inválido")

    if not 1 <= args.end <= 65535:
        raise SystemExit("[!] Puerto final inválido")

    if args.start > args.end:
        raise SystemExit(
            "[!] El puerto inicial debe ser menor"
        )

    try:
        ip = socket.gethostbyname(args.target)
    except socket.gaierror:
        raise SystemExit(
            "[!] No se pudo resolver el objetivo"
        )

    ports = scan(
        ip,
        args.start,
        args.end,
        args.timeout
    )

    report = {
        "tool": "DarkScanner",
        "target": args.target,
        "ip": ip,
        "open_ports": ports,
    }

    with open(
        args.output,
        "w",
        encoding="utf-8"
    ) as f:
        json.dump(
            report,
            f,
            indent=2,
            ensure_ascii=False
        )

    print(f"[+] Puertos abiertos: {len(ports)}")
    print(f"[+] Reporte: {args.output}")
