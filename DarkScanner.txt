import socket
from datetime import datetime

TARGET = input("Objetivo (IP o dominio): ")
START_PORT = 1
END_PORT = 1024

RESULT_FILE = "results/scan.txt"

def scan_port(ip, port):
    try:
        sock = socket.socket()
        sock.settimeout(0.5)
        sock.connect((ip, port))
        return True
    except:
        return False
    finally:
        sock.close()

def main():
    print(f"\n[+] Escaneando {TARGET}...\n")
    inicio = datetime.now()

    abiertos = []

    for port in range(START_PORT, END_PORT + 1):
        if scan_port(TARGET, port):
            print(f"[OPEN] Puerto {port}")
            abiertos.append(port)

    fin = datetime.now()
    tiempo = fin - inicio

    # Guardar resultados
    with open(RESULT_FILE, "w") as f:
        f.write(f"Scan de {TARGET}\n")
        f.write(f"Fecha: {inicio}\n\n")

        for port in abiertos:
            f.write(f"Puerto abierto: {port}\n")

        f.write(f"\nTiempo total: {tiempo}\n")

    print("\n[+] Escaneo terminado")
    print(f"[+] Puertos abiertos: {len(abiertos)}")
    print(f"[+] Resultados guardados en {RESULT_FILE}")
